#### Scatterplot for each patient
PatientCodes = unique(lowperfdata$PatientCode)
for (i in PatientCodes) {
lowperfdataplot = subset(lowperfdata, PatientCode == i)
datacollectionhours = lowperfdataplot$DataCollectionStart[1]/3600
patientcategory = lowperfdataplot$PatientCategory[1]
arrestlocation = lowperfdataplot$ArrestLocation[1]
plot_name = ggplot(lowperfdataplot, aes(x=SecondsBeforeArrest/3600, y=Duration)) + geom_point(color="red", size=1, stroke=0) + theme_classic() + labs(x="Hours Before Arrest", y="Alarm Duration (sec)", title="Low Perfusion Alarms") + annotate("text", x=-36, y=max(lowperfdataplot$Duration)+5, label = paste0("Patient Code: ", i, ", Patient Category: ", patientcategory, ", Hours of Data Collection (Dashed Line): ", round(datacollectionhours, digits=2), ", Arrest Location: ", arrestlocation), size=2) + xlim(-72,0) + geom_vline(xintercept = -datacollectionhours, color="gray", linetype="dashed", linewidth=0.5)
file_name = paste0("plot_",i,".png")
ggsave(filename = file_name, plot = plot_name, width = 5, height = 5, dpi = 300)
}

#### Barplot for each patient summing duration per hour

# create two column matrix with total alarm duration by hour for each patient

PatientCodes = unique(lowperfdata$PatientCode)
for (i in PatientCodes) {
lowperfdataplot = subset(lowperfdata, PatientCode == i)
lowperfdataplot$SecondsBeforeArrest = lowperfdataplot$SecondsBeforeArrest/3600
names(lowperfdataplot)[names(lowperfdataplot) == "SecondsBeforeArrest"] = "HoursBeforeArrest"
twocolumnmatrix = data.frame(matrix(0, nrow=72, ncol=2))
colnames(twocolumnmatrix) = c("Hour", "TotalSeconds")
for (j in 0:71) {
hourcounter = subset(lowperfdataplot, trunc(HoursBeforeArrest) == -j)
twocolumnmatrix[j+1,1] = j
twocolumnmatrix[j+1,2] = sum(hourcounter$Duration)
}
dummy_name = paste0("durationbyhour_",i)
assign(dummy_name, twocolumnmatrix)

# plot duration by hour (still within loop)
patientcategory = lowperfdataplot$PatientCategory[1]
arrestlocation = lowperfdataplot$ArrestLocation[1]
datacollectionhours = lowperfdataplot$DataCollectionStart[1]/3600

plot_name = ggplot(twocolumnmatrix, aes(x=-Hour, y=TotalSeconds)) + geom_col(fill = "red") + theme_classic() + labs(x="Hours Before Arrest", y="Alarm Duration (total seconds per hour)", title="Low Perfusion Alarms: Total Seconds per Hour") + xlim(-72,1) + geom_vline(xintercept = -datacollectionhours, color="gray", linetype="dashed", linewidth=0.5) + annotate("text", x=-36, y=max(twocolumnmatrix$TotalSeconds)+50, label = paste0("Patient Code: ", i, ", Patient Category: ", patientcategory, ", Hours of Data Collection (Dashed Line): ", round(datacollectionhours, digits=2), ", Arrest Location: ", arrestlocation), size=2)
file_name = paste0("plot_",i,"_durationbyhour.png")
ggsave(filename = file_name, plot = plot_name, width = 5, height = 5, dpi = 300)

}
